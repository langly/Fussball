# fussball -- a 2D five-a-side football simulator driven by Python scripts.
#
# Requires Python 3.10 or newer (the code uses dataclasses with slots=True).
# Developed and tested on Python 3.13.14 with pygame-ce 2.5.7.
#
#   python3 -m venv .venv
#   .venv/bin/pip install -r requirements.txt
#
# Only the viewer needs pygame. The simulation, the bot API and the headless
# and batch modes (--headless, --matches) use the standard library alone, so if
# you only ever run matches without a window you can skip this install:
#
#   python3 -m football bots/chaser.py bots/tactician.py --headless
#
# Note: pygame-ce is a drop-in fork of pygame and installs the same `pygame`
# module. Do not install both into one environment -- they will clash.

pygame-ce>=2.4,<3

# The 3D view (--view 3d). The 2D view does not need it, and the 3D view still
# needs pygame-ce, whose pitch drawing it reuses as a ground texture.
panda3d>=1.10,<2

# Optional: lets animated GIFs play on the pitch-side hoardings (--ads).
# Without it, GIFs simply show their first frame and stills work as normal.
pillow>=10,<14

# Development only: the test suite. Not needed to run a match.
#   .venv/bin/pip install pytest && .venv/bin/python -m pytest
pytest>=8,<10
